\begin{frame}{Setup}
\begin{itemize}
  \item \textbf{Model:} TIGER, $d_{\text{model}}=128$, 4 encoder and 4 decoder
        blocks, RQ semantic IDs ($L=4$, width 256), trained from scratch.
  \item \textbf{Attack:} bandwagon, 1\% fake users, one target; Beauty
        (target stratum \texttt{mid} unless stated).
  \item \textbf{Metrics:} SH@10 (spam exposure), N@10 (utility),
        UR $=$ N@10 / own clean N@10,
        removal $= 1 - \text{SH}_{\text{after}}/\text{SH}_{\text{before}}$.
\end{itemize}
\end{frame}

\begin{frame}{Trained-in PKM: the memory collapses}
\begin{itemize}
  \item Trained jointly from the start, every PKM reads the \textbf{same 32 of
        262{,}144 slots} for every token.
  \item Jaccard 1.0 between forget and retain reads, no forget-exclusive slots.
  \item Holds in all 8 encoder and decoder layouts.
  \item The memory outputs a constant; the residual carries the network.
  \item Collapse is a \textbf{routing} problem: giving every value a gradient
        (dense updates) does not change the router's incentive.
\end{itemize}
\end{frame}

\begin{frame}{The switch loss}
Load balancing on the hard top-$k$ assignment, per head and sub-key half
($n = 512$ sub-keys):
\[
  \mathcal{L}_{\text{switch}} = n \sum_i f_i P_i
\]
\begin{itemize}
  \item $f_i$: share of top-$k$ assignments to sub-key $i$ (counted).
  \item $P_i$: mean softmax on \textbf{per-token standardized} scores, so it
        cannot be satisfied by shrinking all scores.
  \item Default now: coef 0.1 per memory (summed over memories) + batchnorm query.
  \item \textbf{Needed throughout:} annealing it away re-collapses (128 slots);
        raw (unstandardized) $P$ also collapses.
\end{itemize}
\end{frame}

\begin{frame}{Item memory: design and settings}
One learnable entry $v_i$ per item, through a linear adapter, \textbf{added to all
SID token embeddings} of that item at the encoder input (MUNKEY-style, item- not
instance-keyed).
\begin{center}
\small
\fit{\begin{tabular}{|l|l|l|}
\hline
arm & what it adds & settings \\
\hline
\texttt{im}     & item memory (input side) & $p_{drop}$ 0.1 (null token), $p_{nbr}$ 0.2, \\
                &                          & 1 to 4 text-embedding neighbors, $\tau$ 0.1 \\
\texttt{imc}    & + consistency KL         & weight 0.1, KL(sg($p_{own}$) $\|$ $p_{nbr}$) \\
\texttt{ims}    & + output score $s_i = \langle h_0, W v_i \rangle$ & loss 1.0, 255 negatives, \\
                & reranks the beams        & rerank weight 1.0 \\
\texttt{imsc}   & score + consistency      & as above \\
\texttt{imfree} & capacity control (new)   & $p_{drop} = p_{nbr} = 0$ \\
\hline
\end{tabular}}
\end{center}
\begin{itemize}
  \item Inference ablations: \texttt{eval\_pathway=null} (memory off),
        \texttt{score\_weight=0} (no rerank), \texttt{input\_term=false} (score only).
  \item Neighbors: frozen text embeddings; neighbor values detached.
\end{itemize}
\end{frame}

\begin{frame}{Takeaways and open questions}
\begin{itemize}
  \item \textbf{PKM post-hoc:} spam removal is a localized decoder retrain;
        PKM is a faster implementation, not the mechanism.
  \item \textbf{Collapse fix:} the standardized switch loss + batchnorm keeps
        trained-in PKMs healthy (47 to 63\% of slots) at no utility cost on
        Beauty; it must stay on throughout training.
  \item \textbf{But} a healthy memory gives no unlearning advantage: the spam
        is not in the slots.
  \item \textbf{Item memory:} \texttt{imsc} beats plain in all 15 poisoned
        Beauty cells (+8 to +34\% N@10), still +8 to +29\% after F+R and NAU, \alert{pending the converged, capacity-controlled
        clean runs}.
  \item Open: Sports cost of the switch loss (1 seed, $-$10\%); whether any
        unlearning algorithm can exploit a separable memory; paper contribution
        (item memory vs.\ collapse fix).
\end{itemize}
\end{frame}
