\begin{frame}{Question}
\begin{itemize}
  \item On poisoned Beauty, \texttt{imsc} beat plain TIGER in all 15 cells
        (+8 to +34\% N@10), but those runs stopped at 4000 steps and
        had more parameters.
  \item \textbf{Now:} on \textbf{clean} data, does the item memory beat plain
        TIGER on test N@10 across seeds, datasets, tokenizers and training
        recipes, or was it luck, training length or capacity?
  \item Data: clean Beauty, Sports, Toys; training seeds 2, 3, 5.
  \item Metric: test N@10 on the checkpoint with the best validation score.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{The four methods}
Shared backbone: TIGER (T5, 4 + 4 layers, 6 heads $\times$ 64,
$d_{\text{model}}=128$, MLP 1024, ReLU), 4 SID tokens per item, beam search.
\begin{center}
\small
\fit{\begin{tabular}{|l|l|l|}
\hline
method & adds & extra loss \\
\hline
\texttt{plain}  & nothing (TIGER) & none \\
\texttt{im}     & item memory, null token + neighbor mixing & none \\
\texttt{imc}    & as \texttt{im} & consistency KL, weight 0.1 \\
\texttt{imfree} & same per-item vectors, no null token, no mixing & none \\
\hline
\end{tabular}}
\end{center}
\begin{itemize}
  \item Extra parameters (\texttt{im}, \texttt{imc}, \texttt{imfree}): one
        128-d vector per item ($\approx$1.5M on Beauty) + a 128$\times$128 adapter.
  \item No output-side reranking in these four arms.
\end{itemize}
\end{frame}

\begin{frame}{Item memory: how it is read}
One learnable vector $v_i$ per catalog item, initialized $\mathcal{N}(0, 0.02)$.
For each history item, $\text{adapter}(v)$ is \textbf{added to all 4 of its SID
token embeddings} before the encoder. During training, per item occurrence:
\[
v = \begin{cases}
  \text{null token (learned)} & \text{w.p. } 0.1 \ \ (p_{drop}) \\
  \sum_j w_j\, \mathrm{sg}(v_j),\ j \in N_{K'}(i) & \text{w.p. } 0.2 \ \ (p_{nbr}) \\
  v_i & \text{otherwise}
\end{cases}
\]
\begin{itemize}
  \item $N_{K'}(i)$: the $K' \sim U\{1..4\}$ nearest items by cosine similarity
        of frozen flan-T5-XL text embeddings; $w = \mathrm{softmax}(\mathrm{sim}/0.1)$.
  \item At evaluation always $v = v_i$.
\end{itemize}
\end{frame}

\begin{frame}{Item memory: the adapter}
$\text{adapter}(v) = W v + b$: \textbf{one} linear layer (128 $\times$ 128 + bias,
16.5k parameters, default PyTorch init), \textbf{shared by all items} (MUNKEY's
$h_\psi$).
\begin{itemize}
  \item Maps the memory vectors into the SID-token embedding space; the table
        $v$ stores per-item content, $W$ decides how it enters the encoder.
  \item Applied to whatever $v$ was drawn (own entry, null token, neighbor mix),
        so it gets a gradient on \emph{every} occurrence, also when the
        neighbor mix itself is detached.
  \item Start: entries $\sim \mathcal{N}(0, 0.02)$ and a zero null token, so
        the memory's first contribution is small.
  \item The bias $b$ is added to every history item, with or without memory:
        memory off (\texttt{eval\_pathway=null}) feeds the null token through
        the adapter (the training-time ``dropped'' input); for \texttt{imfree}
        (null frozen at 0) that is $b$ alone.
\end{itemize}
\end{frame}

\begin{frame}{Item memory: one training step (\texttt{im})}
\begin{enumerate}
  \item Sample a batch of user histories; for every history item draw its
        $v$ (own entry / null token / neighbor mix, as above).
  \item Add $\text{adapter}(v)$ to the item's 4 SID token embeddings; run the
        unchanged encoder and decoder.
  \item Loss: the \textbf{plain TIGER loss only}, cross-entropy on the next
        item's 4 SID tokens. No extra term for \texttt{im}.
  \item One Adam step on everything jointly, from scratch: backbone, $v$
        table, adapter, null token (same lr and schedule as plain).
\end{enumerate}
\begin{itemize}
  \item Gradient reaches $v_i$ only when the own entry was drawn ($\approx$70\%);
        a neighbor mix is detached, so it trains the backbone and adapter
        but not the neighbors' entries.
  \item Only the input side: the target item is never looked up.
\end{itemize}
\end{frame}

\begin{frame}{Item memory: why the two substitutions}
\begin{itemize}
  \item \textbf{Null token (10\%):} the backbone must still predict well
        without the entry, so $v_i$ is a \emph{refinement} on top of the SID
        tokens, not something the model cannot work without.
  \item \textbf{Neighbor mix (20\%):} the backbone sees the average of
        similar items' entries in $v_i$'s place and must predict from it too,
        so $v_i$ stays close to what similar items carry and holds the
        item-specific \emph{residual}.
  \item Together: entries are smooth over the item neighborhood and the
        backbone degrades gracefully when one is missing or replaced.
  \item Without both (\texttt{imfree}): $v_i$ becomes a free per-item
        parameter, overfits early (best val at 3000 to 4000 steps) and gives
        no gain.
\end{itemize}
\end{frame}

\begin{frame}{Consistency loss and the capacity control}
\textbf{Consistency (\texttt{imc}).} A second forward pass replaces every entry
by its full 4-neighbor average (detached):
\[
  \mathcal{L}_{c} = \sum_{h=1}^{4} \mathrm{KL}\big(\mathrm{sg}(p_{\text{own}}^{(h)})
  \,\big\|\, p_{\text{nbr}}^{(h)}\big), \quad \text{weight } 0.1
\]
\begin{itemize}
  \item Trains the backbone so an entry can later be changed or replaced without
        breaking predictions; costs a second forward pass per step.
\end{itemize}
\medskip
\textbf{Capacity control (\texttt{imfree}).} $p_{drop} = p_{nbr} = 0$: same
vectors, always $v_i$.
\begin{itemize}
  \item \texttt{imfree} $\approx$ \texttt{im} $\Rightarrow$ the gain is extra parameters.
  \item \texttt{im} $>$ \texttt{imfree} $\Rightarrow$ the gain is the memory's
        training design.
\end{itemize}
\end{frame}

\begin{frame}{Reading of the results}
\begin{itemize}
  \item \textbf{Item memory helps on both datasets:} +13.5 / +16.1\% (Beauty),
        about +9.5\% (Sports) for \texttt{im} / \texttt{imc}.
  \item \textbf{Training design, not capacity:} \texttt{imfree} loses on both
        ($-$2.4\%, $-$19.5\%).
  \item \textbf{The models rely on the memory:} switched off, \texttt{im} and
        \texttt{imc} fall 23 to 35\%, below plain. \texttt{imc} loses less
        (consistency makes a missing entry less harmful, as intended);
        \texttt{imfree} collapses without its entries.
  \item \texttt{imc} vs.\ \texttt{im}: ahead on Beauty, equal on Sports.
  \item Earlier memory-off numbers were invalid (the memory stayed on:
        \texttt{null} parsed as YAML null); fixed and rerun.
\end{itemize}
\end{frame}
